cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(pico_ball_vault C CXX ASM)
pico_sdk_init()
find_package(Python3 REQUIRED COMPONENTS Interpreter)

add_executable(pico_ball_vault)

# Stamp the build with its git revision (the 'v' console command prints it).
# Evaluated at configure time; "unknown" outside a git checkout.
execute_process(COMMAND git describe --always --dirty --tags
        WORKING_DIRECTORY ${CMAKE_CURRENT_LIST_DIR}
        OUTPUT_VARIABLE PB_GIT_VERSION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT PB_GIT_VERSION)
    set(PB_GIT_VERSION "unknown")
endif()
message(STATUS "pico_ball firmware version: ${PB_GIT_VERSION}")
target_compile_definitions(pico_ball_vault PRIVATE PB_VERSION="${PB_GIT_VERSION}")

pico_generate_pio_header(pico_ball_vault ${CMAKE_CURRENT_LIST_DIR}/spi.pio)
pico_generate_pio_header(pico_ball_vault ${CMAKE_CURRENT_LIST_DIR}/ws2812.pio)

target_include_directories(pico_ball_vault PRIVATE ${CMAKE_CURRENT_LIST_DIR})

target_sources(pico_ball_vault PRIVATE
        main.c
        gb_link.c
        capture.c
        inject.c
        cross_gen.c
        gen_profile.c
        mon_identity.c
        gen3_party.c
        storage.c
        ui.c
        display_console.c # optional Hosyond E32R32P UI over UART1 (GPIO4/5)
        pio_spi.c
        gen3_multiboot.c
        gen3_trade.c
        baked_gen3_mb.c   # generated by tools/gen_baked_gen3.py (gitignored)
        baked_ptgb_mb.c   # generated by tools/gen_baked_ptgb.py (gitignored)
        usb_console.c     # WebUSB-vendor console (custom TinyUSB; CDC optional)
        usb_descriptors.c
        )

# On-device Gen 1/2 -> Gen 3 conversion runs the vendored Pokemon Community
# Conversion Standard (third_party/PCCS, MIT; not redistributed here). It is
# compiled in when tools/gen_pccs_firmware.py has produced pccs_gen/ (scripts/
# setup.sh does that); otherwise a stub is linked and the feature reports itself
# as unavailable.
set(PCCS_DIR ${CMAKE_CURRENT_LIST_DIR}/../../third_party/PCCS)
set(PCCS_GEN ${CMAKE_CURRENT_LIST_DIR}/pccs_gen)
if(EXISTS ${PCCS_GEN}/pccs_tables.c AND EXISTS ${PCCS_DIR}/source/GBPokemon.cpp)
    message(STATUS "PCCS found: building with on-device Gen 1/2 -> Gen 3 conversion")
    target_sources(pico_ball_vault PRIVATE
            pccs_bridge.cpp
            ${PCCS_GEN}/pccs_tables.c
            ${PCCS_DIR}/source/GBPokemon.cpp
            ${PCCS_DIR}/source/Gen1Pokemon.cpp
            ${PCCS_DIR}/source/Gen2Pokemon.cpp
            ${PCCS_DIR}/source/Gen3Pokemon.cpp
            ${PCCS_DIR}/source/Pokemon.cpp
            ${PCCS_DIR}/source/pokemon_data.cpp
            )
    target_include_directories(pico_ball_vault PRIVATE ${PCCS_GEN} ${PCCS_DIR}/include)
    # PCCS is desktop-style C++ (string streams, sizeable locals): the SDK's
    # default 2 KB stack is not enough for it.
    target_compile_definitions(pico_ball_vault PRIVATE PICO_STACK_SIZE=0x4000)
else()
    message(STATUS "PCCS not found: building WITHOUT Gen 1/2 -> Gen 3 conversion")
    target_sources(pico_ball_vault PRIVATE pccs_stub.c)
endif()

target_link_libraries(pico_ball_vault PRIVATE
        pico_stdlib
        hardware_pio
        hardware_flash
        hardware_sync
        hardware_uart
        tinyusb_device     # our own TinyUSB descriptors (vendor, optional CDC) — no stdio_usb
        tinyusb_board
        pico_unique_id
        )

# We provide our own USB stack (usb_console.c + usb_descriptors.c) for a WebUSB
# vendor device, so do NOT enable the SDK's CDC-only stdio_usb (its descriptors
# would conflict). No SDK UART stdio either (display_console.c drives UART1).
pico_enable_stdio_uart(pico_ball_vault 0)

pico_add_extra_outputs(pico_ball_vault)

# The top 128 KiB of RP2040 flash is used as the on-device Pokémon vault
# (storage.c: VAULT_OFFSET). Fail the build if the firmware image grows into
# that reserved region.
add_custom_command(TARGET pico_ball_vault POST_BUILD
        COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:pico_ball_vault>
                ${CMAKE_CURRENT_BINARY_DIR}/pico_ball_vault.sizecheck.bin
        COMMAND ${Python3_EXECUTABLE} -c
                "import os,sys; p='${CMAKE_CURRENT_BINARY_DIR}/pico_ball_vault.sizecheck.bin'; n=os.path.getsize(p); limit=2*1024*1024-128*1024; print(f'Firmware image: {n} bytes (vault starts at {limit})'); sys.exit(0 if n <= limit else 'firmware image overlaps reserved vault flash')"
        VERBATIM)
